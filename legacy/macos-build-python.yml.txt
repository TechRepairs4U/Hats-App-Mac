name: Build macOS App

on:
  push:
    branches: [main]
  workflow_dispatch:

jobs:
  build-macos:
    name: Build ${{ matrix.arch }}
    runs-on: ${{ matrix.runner }}
    strategy:
      fail-fast: false
      matrix:
        include:
          - arch: x64
            runner: macos-26-intel
          - arch: arm64
            runner: macos-26

    steps:
      - name: Check out code
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.14"

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          python -m pip install -r requirements.txt

      - name: Build app bundle
        run: |
          pyinstaller --noconfirm --windowed --name HATS-Installer \
            --icon app_icon.icns \
            --add-data "app_icon.png:." \
            --add-data "app_icon.icns:." \
            --add-data "sdcard_icon.png:." \
            --add-data "bmc_button.png:." \
            hats_installer.py

      - name: Create distributable archives
        run: |
          ditto -c -k --sequesterRsrc --keepParent dist/HATS-Installer.app HATS-Installer-${{ matrix.arch }}.zip
          hdiutil create -volname "HATS Installer" -srcfolder dist/HATS-Installer.app -ov -format UDZO HATS-Installer-${{ matrix.arch }}.dmg

      - name: Upload app zip
        uses: actions/upload-artifact@v4
        with:
          name: HATS-Installer-${{ matrix.arch }}-app
          path: HATS-Installer-${{ matrix.arch }}.zip

      - name: Upload DMG
        uses: actions/upload-artifact@v4
        with:
          name: HATS-Installer-${{ matrix.arch }}-dmg
          path: HATS-Installer-${{ matrix.arch }}.dmg
